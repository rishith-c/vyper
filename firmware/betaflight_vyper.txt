# VYPER Betaflight starting configuration
# 6S / T-Motor V2207 V3 1950KV / HQProp 5x5V1S / F405 / 55A+ ESC / ELRS
#
# This is a reviewed starting point, not a flight tune. Betaflight CLI names
# can change by release and target. Use `diff all` before/after and reject any
# unknown command instead of pasting blindly. PROPS OFF.

mixer QUADX

# The motors are mounted on the aft faces in pusher configuration. Verify each
# physical direction in the Motors tab before setting the mixer convention.
set yaw_motors_reversed = ON

# Use bidirectional DShot only after every ESC channel returns stable telemetry.
set motor_pwm_protocol = DSHOT600
set dshot_bidir = ON
set motor_poles = 14

# ExpressLRS over CRSF.
set serialrx_provider = CRSF
set serialrx_inverted = OFF

# DOGCOM 6S 1380 mAh pack. Calibrate voltage/current against instruments.
set vbat_min_cell_voltage = 330
set vbat_warning_cell_voltage = 350
set vbat_max_cell_voltage = 425
set battery_capacity = 1380

# Link loss removes thrust. GPS Rescue may be configured only after GPS,
# compass-free rescue direction and home-lock behavior are individually tested.
set failsafe_procedure = DROP
set runaway_takeoff_prevention = ON

# Conservative motor limit for first restrained runs. Raise only through the
# validation gates while monitoring current, bus overshoot and temperature.
set motor_output_limit = 50
set dshot_idle_value = 450

set name VYPER
save
